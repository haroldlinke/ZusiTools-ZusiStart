using log4net;
using Sovoma;
using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.Drawing;
using System.IO;
using System.Linq;
using System.Net;
using System.Text;
using System.Text.RegularExpressions;
using System.Threading;
using System.Windows;
using ZusiKlassenLib2;
using ZusiStart.Data;

namespace ZusiStart.Miscellaneous
{
    //=========================================================================
    public class xHttpMiniServer : Disposable
    {
        private const int _NImages = 5;

        private static readonly ILog Log = LogManager.GetLogger(System.Reflection.MethodBase.GetCurrentMethod().DeclaringType);

        private static readonly string[] _suffixesOfForbiddenFiles = new string[]
        {
            ".doc"
        };
        private static readonly string[] _suffixesOfTextFiles = new string[]
        {
            ".js",
            ".css",
            ".htm",
            ".html"
        };

        public static int Port = Properties.Settings.Default.HttpServerPort;

        private readonly MemoryStream[] _defaultImages = new MemoryStream[_NImages];
        private readonly int[] _defaultImageSizes = new int[_NImages];
        private BackgroundWorker _worker;
        private AutoResetEvent _evWorkerCompleted = new(false);
        private readonly Random _random = new();
        private HttpListener _listener;

        //---------------------------------------------------------------------
        public xHttpMiniServer()
        { }

        //---------------------------------------------------------------------
        public void Run()
        {
            if (_worker == null)
            {
                try
                {
                    _listener = new HttpListener();
                    _listener.Prefixes.Add($"http://localhost:{Port}/");
                    _worker = new BackgroundWorker();
                    _evWorkerCompleted.Reset();
                    _worker.WorkerSupportsCancellation = true;
                    _worker.DoWork += MiniServer_DoWork;
                    _worker.RunWorkerAsync(_listener);
                    Log.Debug("minibrowser started");
                }
                catch (Exception ex)
                {
                    Log.Error(ex.ToString());
                }
            }
            else
            {
                Log.Debug("minibrowser is already running");
            }
        }

        //---------------------------------------------------------------------
        protected override void Dispose(bool disposing)
        {
            if (disposing)
            {
                if (_worker != null)
                {
                    _worker.CancelAsync();
                    _listener.Close();
                    _evWorkerCompleted.WaitOne(1000);
                    _worker.Dispose();
                    _worker = null;
                }
                for (int i = 3; i >= 0; i--)
                {
                    _defaultImages[i]?.Dispose();
                    _defaultImages[i] = null;
                }
                _evWorkerCompleted?.Dispose();
                _evWorkerCompleted = null;
            }
        }

        //---------------------------------------------------------------------
        private void MiniServer_DoWork(object sender, DoWorkEventArgs e)
        {
            BackgroundWorker worker = sender as BackgroundWorker;
            byte[] buffer = new byte[16384];
            Stream output = null;
            HttpListener listener = (HttpListener)e.Argument;
            HttpListenerResponse response = null;

            BuildCache();

            listener.Start();

            while (!worker.CancellationPending)
            {
                try
                {
                    HttpListenerContext context = listener.GetContext();
                    if (worker.CancellationPending)
                    {
                        break;
                    }

                    HttpListenerRequest request = context.Request;
                    response = context.Response;
                    output = response.OutputStream;
                    string localUrlPath = request.Url.LocalPath.Replace('/', '\\').ToLower().Substring(1);
                    string localFilePath = Path.Combine(Zusi.DataPath[DataPathType.Official], localUrlPath);
                    if (!File.Exists(localFilePath))
                    {
                        localFilePath = Path.Combine(Zusi.DataPath[DataPathType.DataDir], localUrlPath);
                    }

                    // TODO: response 404, wenn Datei nicht existiert

                    if (request.HttpMethod == "GET")
                    {
                        string ext = Path.GetExtension(localFilePath);
                        if (localUrlPath.StartsWith("dummypage"))
                        {
                            WriteResponse(response, output, Properties.Resources.DummyPage);
                        }
                        else if (localUrlPath.StartsWith("default-") || _suffixesOfForbiddenFiles.Contains(ext))
                        {
                            TimeTableRelation ttr = Application.Current.Dispatcher.Invoke(() => DataManager.Instance.SelectedTimeTableRelation);
                            if (ttr == null)
                            {
                                throw new Exception("Der ausgewählte Fahrplan konnte nicht zugeordnet werden.");
                            }
                            else
                            {
                                DateTime date = ttr.TimeTable.StartTime.Value;
                                Info info = ttr.TimeTable.GetDocument().Info;
                                string html = Properties.Resources.Default.Replace("%EPOCHE%", GetEpoche(date.Year))
                                    .Replace("%EPOCHE_R%", GetEpocheRomanic(date.Year))
                                    .Replace("%DATE%", date.ToString(@"dddd, \den dd.MMMM yyyy ab H U\hr"))
                                    .Replace("%AUTHOR_PLURAL%", info.AutorEintraege.Count != 1 ? "en" : "")
                                    .Replace("%AUTHORS%", GetTimeTableAuthors(info));
                                WriteResponse(response, output, html);
                            }
                        }
                        else
                        {
                            if (localFilePath.EndsWith("_default-image"))
                            {
                                int ix = _random.Next(_NImages);
                                response.ContentLength64 = _defaultImageSizes[ix];
                                _defaultImages[ix].Position = 0;
                                output.Write(_defaultImages[ix].GetBuffer(), 0, _defaultImageSizes[ix]);
                            }
                            else
                            {
                                using FileStream fs = new(localFilePath, FileMode.Open, FileAccess.Read, FileShare.Read);
                                // *.js
                                // *.css
                                // *.htm(l)
                                if (_suffixesOfTextFiles.Contains(ext))
                                {
                                    using StreamReader sr = new(fs);
                                    string s = sr.ReadToEnd();

                                    //
                                    if (ext == ".css" && Path.GetFileNameWithoutExtension(localFilePath) == "timetable_styles")
                                    {
                                        s = s.Replace("background-color: #3D424F", "background-color: #00000000")
                                            .Replace(@"width: auto\9;", @"/*width: auto\9;*/");
                                    }
                                    else if (ext == ".js" && Path.GetFileNameWithoutExtension(localFilePath) == "timetable_scripts")
                                    {
                                        s = s.Replace("if (!hostReachable())", "//if (!hostReachable())");
                                    }

                                    WriteResponse(response, output, s);
                                }
                                else
                                {
                                    response.ContentLength64 = fs.Length;
                                    fs.CopyTo(output);
                                }
                            }
                        }
                    }
                }
                catch (Exception ex)
                {
                    if (!worker.CancellationPending)
                    {
                        Log.Error(ex.ToString());
                        response.ContentLength64 = 0;
                        response.StatusCode = 500;
                        response.StatusDescription = "Internal Server Error";
                        WriteResponse(response, output, Properties.Resources.Err500.Replace("%MESSAGE%", ex.Message));
                    }
                }
                finally
                {
                    output?.Close();
                    output = null;
                }
            }

            Log.Debug("minibrowser finished");
            _evWorkerCompleted.Set();
        }

        //---------------------------------------------------------------------
        private void BuildCache()
        {
            for (int i = 0; i < _NImages; i++)
            {
                _defaultImages[i] = new MemoryStream();
                Bitmap bmp = i switch
                {
                    1 => Properties.Resources.default1,
                    2 => Properties.Resources.default2,
                    3 => Properties.Resources.default3,
                    4 => Properties.Resources.default4,
                    _ => Properties.Resources.default0,
                };
                bmp.Save(_defaultImages[i], System.Drawing.Imaging.ImageFormat.Png);
                _defaultImageSizes[i] = (int)_defaultImages[i].Length;
            }
        }

        //---------------------------------------------------------------------
        private string GetEpoche(int year)
        {
            // I   1835 bis 1920
            // II  1921 bis 1948
            // III 1949 bis 1964
            // IV  1965 bis 1990
            // V   1991 bis 2006
            // VI  2007

            if (year < 1921)
                return "1";
            if (year < 1949)
                return "2";
            if (year < 1965)
                return "3";
            if (year < 1991)
                return "4";
            if (year < 2007)
                return "5";
            else
                return "6";
        }

        //---------------------------------------------------------------------
        private string GetEpocheRomanic(int year)
        {
            if (year < 1835)
                return "?";
            if (year < 1921)
                return "I";
            if (year < 1949)
                return "II";
            if (year < 1965)
                return "III";
            if (year < 1991)
                return "IV";
            if (year < 2007)
                return "V";
            else
                return "VI";
        }

        //---------------------------------------------------------------------
        private string GetTimeTableAuthors(Info info)
        {
            StringBuilder sb = new();
            bool more = false;
            info.AutorEintraege.ForEach(ae =>
            {
                if (more) sb.Append(',');
                sb.Append(' ');
                sb.Append(ae.Name);
                more = true;
            });
            return sb.ToString();
        }

        //---------------------------------------------------------------------
        private void WriteResponse(HttpListenerResponse response, Stream str, string value)
        {
            byte[] buffer = Encoding.UTF8.GetBytes(value);
            response.ContentLength64 = buffer.Length;
            str.Write(buffer, 0, buffer.Length);
        }
    }
}
